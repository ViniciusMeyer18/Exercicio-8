using Ex8;

class Program
{
    static void Main()
    {
        static void ExecutarAtaqueEmLote(List<IAtacante> personagens)
        {
            Console.WriteLine("--- Executando ataque em lote ---");
            foreach (var personagem in personagens)
            {
                personagem.Atacar();
            }
        }

        static void Main()
        {
            var dragao = new Dragao();
            var mago = new Mago();
            var paladino = new Paladino();

            
            List<IAtacante> atacantes = new List<IAtacante> { dragao, mago, paladino };
            ExecutarAtaqueEmLote(atacantes);

            Console.WriteLine();

            
            List<IVoador> voadores = new List<IVoador> { dragao, paladino };
            Console.WriteLine("--- Quem consegue voar ---");
            foreach (var v in voadores)
            {
                v.Voar();
            }

            Console.WriteLine();
            
            List<ICurador> curadores = new List<ICurador> { mago, paladino };
            Console.WriteLine("--- Quem consegue curar ---");
            foreach (var c in curadores)
            {
                c.Curar();
            }
        }
    }
}
